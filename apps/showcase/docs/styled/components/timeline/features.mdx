## Usage

```tsx
import { Timeline } from '@primereact/ui/timeline';
```

```tsx
<Timeline.Root>
    <Timeline.Event>
        <Timeline.Opposite />
        <Timeline.Separator>
            <Timeline.Marker />
            <Timeline.Connector />
        </Timeline.Separator>
        <Timeline.Content />
    </Timeline.Event>
</Timeline.Root>
```
